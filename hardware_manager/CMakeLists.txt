cmake_minimum_required(VERSION 3.8)
project(hardware_manager)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(yaml-cpp REQUIRED)
set(dependencies ament_index_cpp rclcpp rclcpp_action control_msgs trajectory_msgs rcl_interfaces)
foreach(dependency IN LISTS dependencies)
  find_package(${dependency} REQUIRED)
endforeach()
rosidl_generate_interfaces(${PROJECT_NAME} "action/Execute.action")
rosidl_get_typesupport_target(typesupport ${PROJECT_NAME} rosidl_typesupport_cpp)
add_library(hardware_core STATIC
  src/hardware_config.cpp src/motion_library.cpp src/trajectory_builder.cpp
  src/trajectory_controller.cpp src/motion_scheduler.cpp src/motion_executor.cpp
  src/components/hardware_component.cpp)
target_include_directories(hardware_core PUBLIC include)
ament_target_dependencies(hardware_core ${dependencies})
target_link_libraries(hardware_core "${typesupport}" yaml-cpp)
add_executable(hardware_manager_node src/hardware_manager_node.cpp)
target_link_libraries(hardware_manager_node hardware_core)
install(TARGETS hardware_manager_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESIGN_REVIEW.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_gtest(trajectory_builder_test test/trajectory_builder_test.cpp)
  target_link_libraries(trajectory_builder_test hardware_core)
  ament_add_pytest_test(action_integration test/test_action_integration.py TIMEOUT 120)
endif()
ament_export_dependencies(rosidl_default_runtime)
ament_package()
